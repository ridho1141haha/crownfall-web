     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import Link from "next/link";
     5→import {
     6→  Coins,
     7→  Hammer,
     8→  HelpCircle,
     9→  Logs,
    10→  Moon,
    11→  Mountain,
    12→  Pause as PauseIcon,
    13→  Play,
    14→  RotateCcw,
    15→  Save,
    16→  FolderOpen,
    17→  Sun,
    18→  Sunset,
    19→  Target,
    20→  Volume2,
    21→  VolumeX,
    22→  Flame,
    23→  Crown,
    24→  Ship,
    25→  ChevronLeft,
    26→  ChevronRight,
    27→  Hand,
    28→  ArrowUpFromLine,
    29→  Wrench,
    30→  Clock,
    31→  Sparkles,
    32→} from "lucide-react";
    33→import { Button } from "@/components/ui/button";
    34→import { Badge } from "@/components/ui/badge";
    35→import { Progress } from "@/components/ui/progress";
    36→import { Separator } from "@/components/ui/separator";
    37→import { VIEW } from "@/lib/game/data";
    38→import { GameEngine } from "@/lib/game/engine";
    39→import { Renderer } from "@/lib/game/renderer";
    40→import { AudioManager } from "@/lib/game/audio";
    41→import type { Snapshot } from "@/lib/game/types";
    42→
    43→interface Toast {
    44→  id: number;
    45→  text: string;
    46→  born: number;
    47→}
    48→
    49→const PHASE_META = {
    50→  daylight: { label: "Daylight", icon: Sun, tint: "text-amber-300" },
    51→  sunset: { label: "Sunset", icon: Sunset, tint: "text-orange-300" },
    52→  night: { label: "Night", icon: Moon, tint: "text-indigo-200" },
    53→} as const;
    54→
    55→export default function CrownfallGame() {
    56→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
    57→  const engineRef = useRef<GameEngine | null>(null);
    58→  const rendererRef = useRef<Renderer | null>(null);
    59→  const audioRef = useRef<AudioManager | null>(null);
    60→  const toastIdRef = useRef(1);
    61→
    62→  const [snap, setSnap] = useState<Snapshot | null>(null);
    63→  const [screen, setScreen] = useState<"title" | "game">("title");
    64→  const [paused, setPaused] = useState(false);
    65→  const [helpOpen, setHelpOpen] = useState(false);
    66→  const [victory, setVictory] = useState<{ title: string; subtitle: string } | null>(null);
    67→  const [toasts, setToasts] = useState<Toast[]>([]);
    68→  const [muted, setMuted] = useState(false);
    69→
    70→  const pushToast = useCallback((text: string) => {
    71→    setToasts((ts) => {
    72→      const next = [...ts, { id: toastIdRef.current++, text, born: performance.now() }];
    73→      return next.slice(-5);
    74→    });
    75→  }, []);
    76→
    77→  // refs mirroring UI state for stable event handlers
    78→  const screenRef = useRef<"title" | "game">("title");
    79→  const pausedRef = useRef(false);
    80→
    81→  // ---------------------------------------------------------------------------
    82→  // Engine + renderer + main loop
    83→  // ---------------------------------------------------------------------------
    84→  useEffect(() => {
    85→    const audio = new AudioManager();
    86→    audioRef.current = audio;
    87→
    88→    const engine = new GameEngine({
    89→      sfx: (name) => audio.play(name),
    90→      message: (text) => pushToast(text),
    91→      victory: (title, subtitle) => setVictory({ title, subtitle }),
    92→    });
    93→    engineRef.current = engine;
    94→
    95→    const renderer = new Renderer();
    96→    rendererRef.current = renderer;
    97→    void renderer.load();
    98→
    99→    const canvas = canvasRef.current;
   100→    const ctx = canvas?.getContext("2d") ?? null;
   101→
   102→    let raf = 0;
   103→    let last = performance.now();
   104→    let snapAcc = 0;
   105→
   106→    const loop = (now: number) => {
   107→      const dt = Math.min(0.05, (now - last) / 1000);
   108→      last = now;
   109→      engine.update(dt);
   110→      if (ctx && canvas) {
   111→        renderer.render(engine, ctx, VIEW.WIDTH, VIEW.HEIGHT);
   112→      }
   113→      snapAcc += dt;
   114→      if (snapAcc >= 0.1) {
   115→        snapAcc = 0;
   116→        setSnap(engine.getSnapshot());
   117→        const cutoff = now - 4800;
   118→        setToasts((ts) => (ts.length ? ts.filter((t) => t.born >= cutoff) : ts));
   119→      }
   120→      raf = requestAnimationFrame(loop);
   121→    };
   122→    raf = requestAnimationFrame(loop);
   123→
   124→    // --- keyboard ---
   125→    const down = (e: KeyboardEvent) => {
   126→      const eng = engineRef.current;
   127→      if (!eng || screenRef.current !== "game") return;
   128→      audio.unlock();
   129→      const k = e.key.toLowerCase();
   130→      if (["arrowleft", "arrowright", "arrowup", "arrowdown", " "].includes(k)) {
   131→        e.preventDefault();
   132→      }
   133→      if (k === "escape") {
   134→        setPaused((p) => !p);
   135→        return;
   136→      }
   137→      if (pausedRef.current) return;
   138→      if (k === "a" || k === "arrowleft") eng.input.left = true;
   139→      else if (k === "d" || k === "arrowright") eng.input.right = true;
   140→      else if (k === "e") eng.interact();
   141→      else if (k === "q") eng.upgradeAction();
   142→      else if (k === "r") eng.recruitAction();
   143→      else if (k === "b") eng.buildAction();
   144→      else if (k === "f") eng.repairAction();
   145→      else if (k === " ") eng.flareAction();
   146→      else if (k === "t") eng.toggleTimeScale();
   147→      else if (k === "k") eng.manualSave();
   148→      else if (k === "m") setMuted((m) => !m);
   149→      else if (k === "h") setHelpOpen((v) => !v);
   150→    };
   151→    const up = (e: KeyboardEvent) => {
   152→      const eng = engineRef.current;
   153→      if (!eng) return;
   154→      const k = e.key.toLowerCase();
   155→      if (k === "a" || k === "arrowleft") eng.input.left = false;
   156→      else if (k === "d" || k === "arrowright") eng.input.right = false;
   157→    };
   158→    const firstGesture = () => audio.unlock();
   159→    window.addEventListener("keydown", down);
   160→    window.addEventListener("keyup", up);
   161→    window.addEventListener("pointerdown", firstGesture, { once: true });
   162→
   163→    return () => {
   164→      cancelAnimationFrame(raf);
   165→      window.removeEventListener("keydown", down);
   166→      window.removeEventListener("keyup", up);
   167→      window.removeEventListener("pointerdown", firstGesture);
   168→    };
   169→  }, []);
   170→
   171→  // keep refs in sync so the key handler can read them without re-subscribing
   172→  useEffect(() => {
   173→    screenRef.current = screen;
   174→    pausedRef.current = paused || screen !== "game";
   175→    const eng = engineRef.current;
   176→    if (eng) eng.paused = paused || screen !== "game";
   177→    if (paused || screen !== "game") {
   178→      if (eng) {
   179→        eng.input.left = false;
   180→        eng.input.right = false;
   181→      }
   182→    }
   183→  }, [paused, screen]);
   184→
   185→  useEffect(() => {
   186→    audioRef.current?.setMuted(muted);
   187→  }, [muted]);
   188→
   189→  // auto-dismiss victory banner (mirrors the 7s Godot banner)
   190→  useEffect(() => {
   191→    if (!victory) return;
   192→    const t = setTimeout(() => setVictory(null), 8000);
   193→    return () => clearTimeout(t);
   194→  }, [victory]);
   195→
   196→  // ---------------------------------------------------------------------------
   197→  // Actions
   198→  // ---------------------------------------------------------------------------
   199→  const startNew = () => {
   200→    audioRef.current?.unlock();
   201→    engineRef.current?.newGame();
   202→    setScreen("game");
   203→    setPaused(false);
   204→    setVictory(null);
   205→  };
   206→
   207→  const continueGame = () => {
   208→    audioRef.current?.unlock();
   209→    const ok = engineRef.current?.loadGame() ?? false;
   210→    if (ok) {
   211→      setScreen("game");
   212→      setPaused(false);
   213→    } else {
   214→      pushToast("No usable save found — starting a fresh reign.");
   215→      startNew();
   216→    }
   217→  };
   218→
   219→  const saveAndTitle = () => {
   220→    const eng = engineRef.current;
   221→    if (eng) {
   222→      eng.manualSave();
   223→    }
   224→    setPaused(false);
   225→    setScreen("title");
   226→  };
   227→
   228→  const restart = () => {
   229→    engineRef.current?.newGame();
   230→    setPaused(false);
   231→    setVictory(null);
   232→  };
   233→
   234→  const hold = (dir: "left" | "right", value: boolean) => {
   235→    const eng = engineRef.current;
   236→    if (eng) eng.input[dir] = value;
   237→  };
   238→
   239→  const act = (fn: (e: GameEngine) => void) => {
   240→    audioRef.current?.unlock();
   241→    const eng = engineRef.current;
   242→    if (eng && screen === "game" && !paused) fn(eng);
   243→  };
   244→
   245→  const hasSave = snap?.hasSave ?? false;
   246→
   247→  // ---------------------------------------------------------------------------
   248→  // Render
   249→  // ---------------------------------------------------------------------------
   250→  const phaseMeta = snap ? PHASE_META[snap.phase] : PHASE_META.daylight;
   251→  const PhaseIcon = phaseMeta.icon;
   252→
   253→  return (
   254→    <main className="flex min-h-screen flex-col bg-stone-950 text-stone-100">
   255→      <div className="mx-auto w-full max-w-[1400px] flex-1 px-2 py-3 sm:px-4 sm:py-5">
   256→        {/* header strip */}
   257→        <header className="mb-2 flex items-center justify-between gap-2 px-1">
   258→          <div className="flex items-center gap-2">
   259→            <Crown className="h-5 w-5 text-amber-300" aria-hidden />
   260→            <div>
   261→              <h1 className="text-sm font-bold tracking-widest sm:text-base">
   262→                CROWNFALL: THE LAST LIGHT
   263→              </h1>
   264→              <p className="text-[10px] text-stone-400 sm:text-xs">
   265→                Web port of the Godot 4 vertical slice · Forgotten Shore
   266→              </p>
   267→            </div>
   268→          </div>
   269→          {screen === "game" && (
   270→            <div className="flex items-center gap-1.5">
   271→              <Button
   272→                variant="outline"
   273→                size="sm"
   274→                className="border-stone-700 bg-stone-900/90 text-stone-200 hover:bg-stone-800 hover:text-stone-100"
   275→                onClick={() => setHelpOpen(true)}
   276→                aria-label="Help (H)"
   277→              >
   278→                <HelpCircle className="h-4 w-4" />
   279→                <span className="hidden sm:inline">Help</span>
   280→              </Button>
   281→              <Button
   282→                variant="outline"
   283→                size="sm"
   284→                className="border-stone-700 bg-stone-900/90 text-stone-200 hover:bg-stone-800 hover:text-stone-100"
   285→                onClick={() => setMuted((m) => !m)}
   286→                aria-label={muted ? "Unmute (M)" : "Mute (M)"}
   287→              >
   288→                {muted ? <VolumeX className="h-4 w-4" /> : <Volume2 className="h-4 w-4" />}
   289→              </Button>
   290→              <Button
   291→                variant="outline"
   292→                size="sm"
   293→                className="border-stone-700 bg-stone-900/90 text-stone-200 hover:bg-stone-800 hover:text-stone-100"
   294→                onClick={() => setPaused(true)}
   295→                aria-label="Pause (Esc)"
   296→              >
   297→                <PauseIcon className="h-4 w-4" />
   298→                <span className="hidden sm:inline">Menu</span>
   299→              </Button>
   300→            </div>
   301→          )}
   302→        </header>
   303→
   304→        {/* game viewport */}
   305→        <div className="relative w-full overflow-hidden rounded-lg border border-stone-800 bg-black shadow-2xl">
   306→          <canvas
   307→            ref={canvasRef}
   308→            width={VIEW.WIDTH}
   309→            height={VIEW.HEIGHT}
   310→            className="block h-auto w-full [image-rendering:pixelated]"
   311→            aria-label="Crownfall game world"
   312→            role="img"
   313→          />
   314→
   315→          {/* ------- HUD (only during play) ------- */}
   316→          {screen === "game" && snap && (
   317→            <>
   318→              {/* top-left: clock */}
   319→              <div className="pointer-events-none absolute left-1.5 top-1.5 w-40 rounded-md border border-stone-700/70 bg-stone-950/75 p-1.5 backdrop-blur-sm sm:left-2 sm:top-2 sm:w-56 sm:p-2 md:w-64">
   320→                <div className="flex items-center justify-between">
   321→                  <div className="flex items-center gap-1.5">
   322→                    <PhaseIcon className={`h-4 w-4 ${phaseMeta.tint}`} aria-hidden />
   323→                    <span className="text-sm font-semibold">
   324→                      Day {snap.day} · {phaseMeta.label}
   325→                    </span>
   326→                  </div>
   327→                  {snap.timeScale > 1 && (
   328→                    <Badge variant="secondary" className="h-5 px-1.5 text-[10px]">
   329→                      <Clock className="mr-0.5 h-3 w-3" />20x
   330→                    </Badge>
   331→                  )}
   332→                </div>
   333→                <Progress
   334→                  value={snap.phaseProgress * 100}
   335→                  className="mt-1.5 h-1.5"
   336→                  aria-label="Phase progress"
   337→                />
   338→                <div className="mt-1.5 flex items-center gap-2 text-[11px] text-stone-300">
   339→                  <span className="inline-flex items-center gap-0.5">
   340→                    <Coins className="h-3 w-3 text-amber-300" /> {snap.gold}
   341→                  </span>
   342→                  <span className="inline-flex items-center gap-0.5">
   343→                    <Logs className="h-3 w-3 text-orange-200" /> {snap.wood}
   344→                  </span>
   345→                  <span className="inline-flex items-center gap-0.5">
   346→                    <Mountain className="h-3 w-3 text-stone-300" /> {snap.stone}
   347→                  </span>
   348→                  <span className="ml-auto inline-flex items-center gap-0.5">
   349→                    <Flame className="h-3 w-3 text-orange-300" /> Lv.{snap.hearthLevel}
   350→                  </span>
   351→                </div>
   352→              </div>
   353→
   354→              {/* top-right: forecast / alert + population */}
   355→              <div className="pointer-events-none absolute right-1.5 top-1.5 w-44 rounded-md border border-stone-700/70 bg-stone-950/75 p-1.5 text-right text-[10px] backdrop-blur-sm sm:right-2 sm:top-2 sm:w-72 sm:p-2 sm:text-[11px] md:w-80">
   356→                {snap.alert ? (
   357→                  <p className="font-semibold text-red-300">{snap.alert}</p>
   358→                ) : (
   359→                  <p className="text-sky-200/90">{snap.forecast}</p>
   360→                )}
   361→                <p className="mt-1 text-stone-400">{snap.population}</p>
   362→                {(snap.flareActive || snap.flareCooldown > 0) && (
   363→                  <p className="mt-1 text-orange-300">
   364→                    {snap.flareActive
   365→                      ? "KINDLE FLARE ACTIVE (+50% light damage)"
   366→                      : `Flare cooldown ${Math.ceil(snap.flareCooldown)}s`}
   367→                  </p>
   368→                )}
   369→                {snap.hearthBoost > 0 && (
   370→                  <p className="mt-1 text-orange-200">
   371→                    Hearth stoked ({Math.ceil(snap.hearthBoost)}s)
   372→                  </p>
   373→                )}
   374→                {snap.reign === "ISLAND_CLEARED" && (
   375→                  <p className="mt-1 font-semibold text-emerald-300">
   376→                    Island cleared — the kingdom is at peace
   377→                  </p>
   378→                )}
   379→              </div>
   380→
   381→              {/* bottom-center: context prompt */}
   382→              {snap.prompt && (
   383→                <div className="pointer-events-none absolute bottom-16 left-1/2 -translate-x-1/2">
   384→                  <div className="flex items-center gap-2 rounded-full border border-amber-300/40 bg-stone-950/85 px-4 py-1.5 text-xs text-amber-200 shadow-lg backdrop-blur">
   385→                    <kbd className="rounded border border-amber-300/50 bg-stone-900 px-1.5 py-0.5 font-mono text-[11px] text-amber-200">
   386→                      {snap.promptKey}
   387→                    </kbd>
   388→                    <span>{snap.prompt}</span>
   389→                  </div>
   390→                </div>
   391→              )}
   392→
   393→              {/* bottom-left: feedback toasts */}
   394→              <div className="pointer-events-none absolute bottom-2 left-2 flex w-80 flex-col gap-1">
   395→                {toasts.slice(-3).map((t) => (
   396→                  <div
   397→                    key={t.id}
   398→                    className="rounded border border-stone-700/60 bg-stone-950/80 px-2 py-1 text-[11px] text-stone-200 backdrop-blur-sm"
   399→                  >
   400→                    {t.text}
   401→                  </div>
   402→                ))}
   403→              </div>
   404→
   405→              {/* desktop controls hint */}
   406→              <div className="pointer-events-none absolute bottom-2 right-2 hidden rounded-md border border-stone-700/60 bg-stone-950/70 px-2 py-1 text-[10px] text-stone-400 backdrop-blur-sm lg:block">
   407→                A/D move · E interact · Q upgrade · R recruit · B build · F repair ·
   408→                Space flare · T 20x · K save · M mute · Esc menu
   409→              </div>
   410→
   411→              {/* touch controls */}
   412→              <div className="absolute bottom-2 left-1/2 flex -translate-x-1/2 flex-nowrap items-center justify-center gap-1 lg:hidden">
   413→                <TouchButton
   414→                  label="Left"
   415→                  onPress={(v) => hold("left", v)}
   416→                  ariaLabel="Move left"
   417→                >
   418→                  <ChevronLeft className="h-5 w-5" />
   419→                </TouchButton>
   420→                <TouchButton
   421→                  label="Right"
   422→                  onPress={(v) => hold("right", v)}
   423→                  ariaLabel="Move right"
   424→                >
   425→                  <ChevronRight className="h-5 w-5" />
   426→                </TouchButton>
   427→                <ActionButton icon={Hand} kbd="E" onPress={() => act((e) => e.interact())} ariaLabel="Interact" />
   428→                <ActionButton icon={ArrowUpFromLine} kbd="Q" onPress={() => act((e) => e.upgradeAction())} ariaLabel="Upgrade" />
   429→                <ActionButton icon={Sparkles} kbd="R" onPress={() => act((e) => e.recruitAction())} ariaLabel="Recruit" />
   430→                <ActionButton icon={Hammer} kbd="B" onPress={() => act((e) => e.buildAction())} ariaLabel="Build" />
   431→                <ActionButton icon={Wrench} kbd="F" onPress={() => act((e) => e.repairAction())} ariaLabel="Repair" />
   432→                <ActionButton icon={Flame} kbd="␣" onPress={() => act((e) => e.flareAction())} ariaLabel="Kindle flare" />
   433→              </div>
   434→            </>
   435→          )}
   436→
   437→          {/* ------- Title screen ------- */}
   438→          {screen === "title" && (
   439→            <div className="absolute inset-0 flex items-center justify-center bg-gradient-to-b from-stone-950/85 via-stone-950/70 to-stone-950/90 backdrop-blur-[2px]">
   440→              <div className="w-full max-w-md px-6 text-center">
   441→                <Crown className="mx-auto h-10 w-10 text-amber-300" aria-hidden />
   442→                <h2 className="mt-3 text-3xl font-black tracking-widest text-amber-100">
   443→                  CROWNFALL
   444→                </h2>
   445→                <p className="text-sm font-semibold tracking-[0.3em] text-stone-400">
   446→                  THE LAST LIGHT
   447→                </p>
   448→                <p className="mt-3 text-xs leading-relaxed text-stone-300">
   449→                  Build a kingdom. Explore forgotten lands. Survive the darkness.
   450→                  Collect Gold, Wood &amp; Stone by day — hold the wall against the
   451→                  Greedling raiders by night.
   452→                </p>
   453→                <div className="mt-5 flex flex-col gap-2">
   454→                  <Button onClick={startNew} size="lg" className="gap-2">
   455→                    <Play className="h-4 w-4" /> New Reign
   456→                  </Button>
   457→                  <Button
   458→                    onClick={continueGame}
   459→                    size="lg"
   460→                    variant="secondary"
   461→                    className="gap-2"
   462→                    disabled={!hasSave}
   463→                  >
   464→                    <FolderOpen className="h-4 w-4" />
   465→                    {hasSave ? "Continue Reign" : "No Save Found"}
   466→                  </Button>
   467→                  <Button
   468→                    onClick={() => setHelpOpen(true)}
   469→                    size="lg"
   470→                    variant="outline"
   471→                    className="gap-2 border-stone-700 bg-stone-900/90 text-stone-200 hover:bg-stone-800 hover:text-stone-100"
   472→                  >
   473→                    <HelpCircle className="h-4 w-4" /> How to Play
   474→                  </Button>
   475→                </div>
   476→                <p className="mt-4 text-[10px] text-stone-500">
   477→                  Faithful browser port of{" "}
   478→                  <Link
   479→                    href="https://github.com/ridho1141haha/crownfall"
   480→                    target="_blank"
   481→                    rel="noreferrer"
   482→                    className="underline hover:text-stone-300"
   483→                  >
   484→                    github.com/ridho1141haha/crownfall
   485→                  </Link>{" "}
   486→                  (Godot 4 · GDScript) using its original assets, data files and
   487→                  system rules.
   488→                </p>
   489→              </div>
   490→            </div>
   491→          )}
   492→
   493→          {/* ------- Pause menu ------- */}
   494→          {screen === "game" && paused && (
   495→            <div className="absolute inset-0 flex items-center justify-center bg-stone-950/80 backdrop-blur-sm">
   496→              <div className="w-full max-w-xs rounded-lg border border-stone-700 bg-stone-900/95 p-5 shadow-2xl">
   497→                <h3 className="text-center text-lg font-bold tracking-widest text-amber-100">
   498→                  PAUSED
   499→                </h3>
   500→                <p className="mt-1 text-center text-[11px] text-stone-400">
   501→                  {snap ? `Day ${snap.day} · ${phaseMeta.label} · ${snap.gold}G ${snap.wood}W ${snap.stone}S` : ""}
   502→                </p>
   503→                <div className="mt-4 flex flex-col gap-2">
   504→                  <Button onClick={() => setPaused(false)} className="gap-2">
   505→                    <Play className="h-4 w-4" /> Resume
   506→                  </Button>
   507→                  <Button
   508→                    variant="secondary"
   509→                    className="gap-2"
   510→                    onClick={() => {
   511→                      const ok = engineRef.current?.manualSave() ?? false;
   512→                      pushToast(ok ? "Kingdom state saved." : "Save failed (disabled during Night).");
   513→                    }}
   514→                  >
   515→                    <Save className="h-4 w-4" /> Save (K)
   516→                  </Button>
   517→                  <Button
   518→                    variant="secondary"
   519→                    className="gap-2"
   520→                    onClick={() => {
   521→                      const ok = engineRef.current?.loadGame() ?? false;

   658→function TouchButton({
   659→  label,
   660→  onPress,
   661→  ariaLabel,
   662→  children,
   663→}: {
   664→  label: string;
   665→  onPress: (down: boolean) => void;
   666→  ariaLabel: string;
   667→  children: React.ReactNode;
   668→}) {
   669→  return (
   670→    <button
   671→      type="button"
   672→      aria-label={ariaLabel}
   673→      className="flex h-10 w-10 select-none items-center justify-center rounded-full border border-stone-600 bg-stone-900/85 text-stone-100 shadow-md active:bg-stone-800"
   674→      onPointerDown={(e) => {
   675→        e.preventDefault();
   676→        onPress(true);
   677→      }}
   678→      onPointerUp={() => onPress(false)}
   679→      onPointerLeave={() => onPress(false)}
   680→      onPointerCancel={() => onPress(false)}
   681→      onContextMenu={(e) => e.preventDefault()}
   682→    >
   683→      {children}
   684→      <span className="sr-only">{label}</span>
   685→    </button>
   686→  );
   687→}
